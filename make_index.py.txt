function unformatNumber(v){
  if (v === null || v === undefined) return 0;
  var s = String(v).replace(/[^\d\-]/g, '');
  var n = Number(s);
  return isNaN(n) ? 0 : n;
}
function formatThousandInput(el){
  var raw = el.value || '';
  var isNeg = raw.trim().startsWith('-');
  var digits = raw.replace(/[^\d]/g, '');
  if (digits === ''){ el.value = isNeg ? '-' : ''; return; }
  var num = parseInt(digits, 10);
  el.value = (isNeg ? '-' : '') + num.toLocaleString('en-US');
}

function applyColSearch(){
  var rows = _lastReportRows;
  var keys = Object.keys(_colSearchValues);
  if (keys.length){
    rows = rows.filter(function(r){
      for (var i = 0; i < keys.length; i++){
        var k = keys[i];
        var target = r[k];
        var s = (target === null || target === undefined) ? '' : String(target).toLowerCase();
        if (s.indexOf(_colSearchValues[k]) === -1) return false;
      }
      return true;
    });
  }
  renderReportRows(rows);
}

function renderBars(containerId, data, opts){
  opts = opts || {};
  var maxV = opts.max || Math.max.apply(null, data.map(function(x){return x.value;}).concat([1]));
  var colorCls = opts.color || '';
  var el = document.getElementById(containerId);
  if (!el) return;
  if (!data.length){
    el.innerHTML = '<div style="color:var(--text-3);font-size:12px;text-align:center;padding:24px">داده‌ای ثبت نشده است</div>';
    return;
  }
  el.innerHTML = data.map(function(x){
    var pct = maxV > 0 ? (x.value / maxV) * 100 : 0;
    var cls = x.color || colorCls;
    return '<div class="bar-row">' +
      '<div class="name" title="' + (x.label || '') + '">' + (x.label || '') + '</div>' +
      '<div class="bar-wrap"><div class="bar ' + cls + '" style="width:' + pct + '%"></div></div>' +
      '<div class="val">' + (opts.formatter ? opts.formatter(x.value) : fmt(x.value)) + '</div>' +
    '</div>';
  }).join('');
}

function renderDonut(containerId, parts, centerLabel){
  var total = parts.reduce(function(s, p){ return s + p.value; }, 0);
  var el = document.getElementById(containerId);
  if (!el) return;
  if (total <= 0){
    el.innerHTML = '<div style="color:var(--text-3);font-size:12px;text-align:center;padding:24px">داده‌ای ثبت نشده است</div>';
    return;
  }
  var stops = [];
  var acc = 0;
  parts.forEach(function(p){
    var start = (acc / total) * 100;
    acc += p.value;
    var end = (acc / total) * 100;
    stops.push(p.color + ' ' + start + '% ' + end + '%');
  });
  var donut = '<div class="donut" style="background:conic-gradient(from -90deg, ' + stops.join(',') + ')">' +
    '<div class="donut-center"><div class="num">' + fmt(total) + '</div>' +
    '<div class="lbl">' + (centerLabel || 'کل') + '</div></div></div>';
  var legend = '<div class="donut-legend">' + parts.map(function(p){
    var pct = ((p.value / total) * 100).toFixed(1);
    return '<div class="item">' +
      '<span class="dot" style="background:' + p.color + '"></span>' +
      '<span class="lbl">' + p.label + '</span>' +
      '<span class="v">' + fmt(p.value) + ' (' + pct + '٪)</span>' +
    '</div>';
  }).join('') + '</div>';
  el.innerHTML = donut + legend;
}

function buildMultiSelect(containerId, options) {
  var c = document.getElementById(containerId);
  if (!c) return;
  var html = '<div class="ms-header" onclick="toggleMs(\''+containerId+'\')"><span id="'+containerId+'-label">همه</span><span>▼</span></div><div class="ms-dropdown" id="'+containerId+'-dropdown"><label class="ms-item"><input type="checkbox" value="همه" checked onchange="onMsChange(\''+containerId+'\',this)"> <span>همه</span></label>';
  options.forEach(function(opt) {
    if (opt === 'همه') return;
    html += '<label class="ms-item"><input type="checkbox" value="'+opt+'" onchange="onMsChange(\''+containerId+'\',this)"> <span>'+opt+'</span></label>';
  });

  c.innerHTML = html;
}
function toggleMs(id) {
  var dd = document.getElementById(id + '-dropdown');
  document.querySelectorAll('.ms-dropdown').forEach(function(el) { if (el.id !== id + '-dropdown') el.classList.remove('show'); });
  dd.classList.toggle('show');
}
function onMsChange(containerId, cb) {
  var dd = document.getElementById(containerId + '-dropdown');
  var allCb = dd.querySelector('input[value="همه"]');
  var others = dd.querySelectorAll('input:not([value="همه"])');
  if (cb.value === 'همه' && cb.checked) { others.forEach(function(x){x.checked=false;}); }
  else if (cb.value !== 'همه' && cb.checked) { allCb.checked = false; }
  var checked = Array.from(dd.querySelectorAll('input:checked')).map(function(x){return x.value;});
  if (checked.length === 0) { allCb.checked = true; checked = ['همه']; }
  var label = document.getElementById(containerId + '-label');
  if (checked.includes('همه')) label.textContent = 'همه';
  else if (checked.length === 1) label.textContent = checked[0];
  else label.textContent = checked.length + ' مورد انتخاب شده';
  if (window._reportTimer) clearTimeout(window._reportTimer);
  window._reportTimer = setTimeout(loadReport, 300);
}
function getSelectedValues(containerId) {
  var dd = document.getElementById(containerId + '-dropdown');
  if (!dd) return ['همه'];
  var checked = Array.from(dd.querySelectorAll('input:checked')).map(function(x){return x.value;});
  if (checked.length === 0 || checked.includes('همه')) return ['همه'];
  return checked;
}

async function doBulkImport(category){
  var fileInput = document.getElementById('bulk-' + category + '-file');
  var msg = document.getElementById('bulk-' + category + '-msg');

  if (!fileInput.files.length){
    msg.innerHTML = '<span style="color:#fca5a5">فایل را انتخاب کنید.</span>';
    return;
  }

  var fd = new FormData();
  fd.append('file', fileInput.files[0]);

  msg.innerHTML = '<span style="color:#a1a1aa">در حال ایمپورت...</span>';

  try {
    var res = await fetch('/api/bulk-import/' + category, {method: 'POST', body: fd});
    var data = await res.json();

    if (!res.ok){
      msg.innerHTML = '<span style="color:#fca5a5">' + (data.error || 'خطا') + '</span>';
      return;
    }

    var r = data.result;
    var txt = r.centers_count + ' مرکز — درج: ' + r.inserted + '، به‌روز: ' + r.updated;

    if (r.remapped_count > 0){
      txt += ' — تطبیق هوشمند: ' + r.remapped_count;
    }
    if (r.unmatched_count > 0){
      txt += ' — ⚠️ ناشناخته: ' + r.unmatched_count;
    }

    msg.innerHTML = '<span style="color:#6ee7b7">' + txt + '</span>';

    fileInput.value = '';

    loadSpecialists();

    // نمایش مراکز ناشناخته
    if (r.unmatched && r.unmatched.length){
      showUnmatched(r.unmatched, category);
    } else {
      document.getElementById('unmatched-box').style.display = 'none';
    }
  } catch(e){
    msg.innerHTML = '<span style="color:#fca5a5">خطا: ' + e + '</span>';
  }
}

async function applyRemap(idx, items, category){
  var sel = document.querySelector('.unmatched-target[data-idx="' + idx + '"]');
  var target = sel ? sel.value : '';
  if (!target){
    alert('مرکز صحیح را انتخاب کنید');
    return;
  }
  var it = items[idx];
  var res = await fetch('/api/bulk-import/remap', {
    method: 'POST',
    headers: {'Content-Type': 'application/json'},
    body: JSON.stringify({
      source: it.raw_name,
      target: target,
      amount: it.amount,
      category: category
    })
  });
  var d = await res.json();
  if (res.ok){
    showToast('تخصیص انجام شد', it.raw_name + ' → ' + target, 'ok');
    // حذف سطر
    var tr = document.querySelector('#unmatched-table tr[data-idx="' + idx + '"]');
    if (tr) tr.remove();
    // اگه خالی شد، مخفی کن
    var tbody = document.querySelector('#unmatched-table tbody');
    if (tbody && !tbody.querySelector('tr')){
      document.getElementById('unmatched-box').style.display = 'none';
    }
    loadSpecialists();
  } else {
    alert(d.error || 'خطا');
  }
}

async function saveAllManualCeilings(){
  var period = getPeriod('mc-year', 'mc-month');
  var inputs = document.querySelectorAll('#manual-ceilings-table input.mc-amount');
  var saved = 0;
  for (var i = 0; i < inputs.length; i++){
    var inp = inputs[i];
    var idx = Number(inp.dataset.idx);
    var row = _mcCache[idx];
    if (!row) continue;
    var val = inp.value;
    if (val === '' || val === null) continue;
    var num = unformatNumber(val);
    if (isNaN(num)) continue;
    var res = await fetch('/api/manual-ceilings/save', {
      method: 'POST', headers: {'Content-Type':'application/json'},
      body: JSON.stringify({
        center_name: row.center_name,
        supply_type: row.supply_type || '',
        period_month: period,
        amount: num
      })
    });
    if (res.ok) saved++;
  }
  var msg = document.getElementById('mc-msg');
  msg.innerHTML = '<div class="msg ok">'+saved+' سقف دستی ثبت شد.</div>';
  loadManualCeilings();
  setTimeout(function(){ msg.innerHTML=''; }, 3000);
}
async function deleteManualCeiling(id){
  if (!confirm('این سقف دستی حذف شود؟')) return;
  await fetch('/api/manual-ceilings/' + id, {method: 'DELETE'});
  loadManualCeilings();
}

async function clearMonthData(type){
  var period, label, payload;
  if (type === 'records'){
    period = getPeriod('period-year', 'period-month');
    label = 'اکسل ورودی';
    payload = { period_month: period, clear_records: true };
  } else if (type === 'employment'){
    period = getPeriod('emp-year', 'emp-month');
    label = 'وضعیت اشتغال';
    payload = { period_month: period, clear_employment: true };
  } else { return; }
  var msg = 'آیا مطمئن هستید که می‌خواهید داده‌های «' + label + '» برای دورهٔ «' + period + '» را پاک کنید؟\n\n';

  if (!confirm(msg)) return;
  try {
    var res = await fetch('/api/clear-month', {
      method: 'POST',
      headers: {'Content-Type': 'application/json'},
      body: JSON.stringify(payload)
    });
    var data = await res.json();
    if (res.ok){
      var d = data.deleted || {};
      var parts = [];

    } else {
      alert('خطا: ' + (data.error || 'نامعلوم'));

  } catch (e) {
    alert('خطا: ' + e);
  }
}

function renderApprovalDeputies(rows){

      '<td class="num">'+fmt(Math.round(r.pending_amount))+'</td>' +

async function doImportMainCenter(){
  var fileInput = document.getElementById('maincenter-file');
  var msg = document.getElementById('maincenter-msg');

  if (!fileInput.files.length){
    msg.innerHTML = '<div class="msg err">فایل را انتخاب کنید.</div>';
    return;
  }

  var fd = new FormData();
  fd.append('file', fileInput.files[0]);

  msg.innerHTML = '<div class="msg">در حال بارگذاری...</div>';

  try {
    var res = await fetch('/api/import-main-center', {method: 'POST', body: fd});
    var data = await res.json();

    if (!res.ok){
      msg.innerHTML = '<div class="msg err">' + (data.error || 'خطا') + '</div>';
      return;
    }

    var r = data.import_result;
    var html = '<div class="msg ok">' + r.ok + ' ردیف خوانده شد';
    html += '<br>' + r.records_updated + ' رکورد به‌روزرسانی شد';
    if (r.skipped > 0) html += '<br>' + r.skipped + ' ردیف رد شد';
    html += '</div>';
    msg.innerHTML = html;

    fileInput.value = '';
  } catch(e){
    msg.innerHTML = '<div class="msg err">خطا: ' + e + '</div>';
  }
}

function openHRView(view){
  // بستن dropdown
  var dd = document.getElementById('hr-dropdown');
  if (dd) dd.style.display = 'none';

  // فعال کردن تَب
  document.querySelectorAll('.tab').forEach(function(b){b.classList.remove('active');});
  var tab = document.getElementById('hr-tab');
  if (tab) tab.classList.add('active');
  document.querySelectorAll('main > section').forEach(function(s){s.classList.add('hidden');});
  var sec = document.getElementById('tab-personnel');
  if (sec) sec.classList.remove('hidden');
  currentTab = 'personnel';

  // علامت‌گذاری آیتم فعال
  document.querySelectorAll('.hr-dd-item').forEach(function(el){
    el.classList.toggle('active', el.dataset.hrView === view);
  });

  // بارگذاری iframe
  var ifr = document.getElementById('personnel-frame');
  if (!ifr) return;

  if (ifr.dataset.loaded === '0' || !ifr.src){
    ifr.src = '/personnel?embedded=1&view=' + view;
    ifr.dataset.loaded = '1';
  } else {
    // قبلاً لود شده — postMessage
    try {
      ifr.contentWindow.postMessage({action: 'openHRView', view: view}, '*');
    } catch(e){
      // اگه postMessage کار نکرد، src رو عوض کن
      ifr.src = '/personnel?embedded=1&view=' + view;
    }
  }
}
